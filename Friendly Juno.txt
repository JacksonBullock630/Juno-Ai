        self.prohibited_words = [
            "nsfw", "inappropriate", "offensive", "hate", "violence", 
            "abuse", "racist", "sexist", "explicit", "harassment", 
            "threat", "profanity", "slur", "discrimination"
        ]  # Expanded prohibited words list

    def load_juno_data(self):
        if os.path.exists(JUNO_DATA_FILE):
            with open(JUNO_DATA_FILE, "r") as file:
                return file.read().splitlines()
        return []

    def load_user_data(self):
        if os.path.exists(USER_DATA_FILE):
            with open(USER_DATA_FILE, "r") as file:
                return json.load(file)
        return {}

    def load_math_data(self):
        if os.path.exists(MATH_JUNO_FILE):
            with open(MATH_JUNO_FILE, "r") as file:
                return file.read().splitlines()
        return []

    def save_user_data(self):
        with open(USER_DATA_FILE, "w") as file:
            json.dump(self.user_data, file)

    def log_chat(self, message):
        with open(CHAT_HISTORY_FILE, "a") as file:
            file.write(message + "\n")

    def greet_user(self):
        if self.user_name in self.user_data:
            print(f"Welcome back, {self.user_name}!")
        else:
            print("Hello! I don't think we've met before. What's your name?")
            self.user_name = input("Your name: ").strip()
            self.user_data[self.user_name] = {}
            self.save_user_data()
            print(f"Nice to meet you, {self.user_name}!")

    def solve_math(self, question):
        # Simple math-solving logic using math_data
        for line in self.math_data:
            if question.lower() in line.lower():
                return f"Here's what I found: {line}"
        try:
            # Attempt to evaluate the math expression
            result = eval(question)
            return f"The answer is: {result}"
        except Exception:
            return "I'm not sure how to solve that. Can you clarify?"

    def is_inappropriate(self, user_input):
        # Check if the input contains prohibited words
        for word in self.prohibited_words:
            if word in user_input.lower():
                return True
        return False

    def fuzzy_match(self, user_input):
        # Use difflib to find the closest match in default_responses
        closest_match = difflib.get_close_matches(user_input.lower(), self.default_responses.keys(), n=1, cutoff=0.6)
        if closest_match:
            return self.default_responses[closest_match[0]]
        return None

    def respond(self, user_input):
        # Check for inappropriate content
        if self.is_inappropriate(user_input):
            return "I'm sorry, but I cannot respond to that."

        # Check for predefined responses with fuzzy matching
        response = self.fuzzy_match(user_input)
        if response:
            return response

        # Check for math-related queries
        if any(char.isdigit() for char in user_input) or any(op in user_input for op in "+-*/="):
            return self.solve_math(user_input)

        # Simple response logic using Juno data
        for line in self.juno_data:
            if user_input.lower() in line.lower():
                return f"Based on what I know: {line}"
        return "I'm not sure about that. Can you tell me more?"

    def chat(self):
        print("Hi! I'm Juno, your friendly chatbot. Type 'exit' to end the chat.")
        self.user_name = input("Please enter your name: ").strip()
        self.greet_user()

        while True:
            user_input = input(f"{self.user_name}: ").strip()
            if user_input.lower() == "exit":
                print("Goodbye! Chat history saved.")
                break
            self.log_chat(f"{self.user_name}: {user_input}")
            response = self.respond(user_input)
            print(f"Juno: {response}")
